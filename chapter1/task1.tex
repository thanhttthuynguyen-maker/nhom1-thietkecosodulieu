Không lưu
\begin{itemize}

    \item \textbf{Toàn bộ source code:}
    chỉ lưu path + commit hash + embedding vector
    (vào vector store riêng), không copy nguyên file vào DB nghiệp vụ.

    \item \textbf{Prompt/response thô của LLM:}
    chỉ giữ lại grounding evidence (đã trích) + confidence score;
    log đầy đủ (nếu cần) đưa vào cold storage/log store riêng,
    có TTL, không nằm trong schema domain.

    \item \textbf{Access token/refresh token OAuth:}
    lưu ở kho secret riêng (mã hóa), không phải là một
    ``thực thể nghiệp vụ'' trong data dictionary.

    \item \textbf{Số liệu dashboard/health metrics tính realtime:}
    tính từ dữ liệu gốc lúc query hoặc cache Redis,
    không phải nguồn sự thật (không cần bảng riêng).

    \item \textbf{Kafka event payload:}
    là log hạ tầng (event sourcing ngắn hạn), không thuộc schema domain,
    có retention riêng.

    \item \textbf{Trạng thái UI tạm thời:}
    tab đang mở, cursor phân trang, \ldots{} không lưu.

    \item \textbf{System health/infra metrics:}
    đẩy sang hệ thống monitoring, không vào DB nghiệp vụ.

\end{itemize}

\section{Phân loại Tác nhân \& Phạm vi Nghiệp vụ (Actor Classification \& Business Scope)}
\label{sec:actor-classification}

\subsection{Danh mục Tác nhân Hệ thống (System Actors)}
\label{sec:system-actors}

Trong mô hình thiết kế quan niệm, hệ thống ghi nhận 5 nhóm
\textbf{Tác nhân (Actors)}. Tất cả các tác nhân con người đều thuộc thực
thể trung tâm User, tác động lên dữ liệu thông qua mô hình phân định vai trò
(Role) có phạm vi (Scoped) gắn liền với thực thể Workspace. Một User có thể nắm
giữ các vai trò khác nhau trên từng Workspace khác nhau.

\subsubsection{Developer}

\textbf{Mục tiêu:} người tạo code và là đối tượng chính tiêu thụ +
khởi tạo tài liệu.

\textbf{Quyền hạn chính:}

\begin{itemize}
    \item Quản lý tài khoản: đăng ký/đăng nhập/đăng xuất
    (email hoặc GitHub OAuth), profile, notification preferences.

    \item Kết nối/ngắt kết nối GitHub, cấp quyền OAuth, chọn repo vào
    workspace, trigger sync thủ công.

    \item Yêu cầu AI sinh tài liệu (theo file/module/repo), chọn template,
    xem coverage gap, preview/chỉnh sửa tài liệu AI, xem grounding evidence,
    submit vào review.

    \item Bật/tắt auto-update tài liệu; xem diff, so sánh version,
    accept/edit/dismiss/regenerate/rollback.

    \item Xem drift events liên quan đến PR của mình
    (loại drift, severity, evidence).

    \item Dùng Knowledge Base: tìm doc/code entity, xem liên kết
    Doc $\leftrightarrow$ Code.
\end{itemize}

\textbf{Phạm vi dữ liệu:} chỉ trong workspace/repo mình được cấp quyền;
\textbf{không} có quyền approve/publish -- chỉ submit và chờ Staff/Manager.

\textbf{Đặc điểm quan trọng:} đây là actor duy nhất
\textbf{khởi tạo} vòng đời tài liệu (bước đầu tiên trong chuỗi
AI $\rightarrow$ Staff $\rightarrow$ Manager).


\subsubsection{Staff}

\textbf{Mục tiêu:} review cấp 1 -- kiểm tra tính đúng đắn nội dung AI
sinh ra trước khi lên Manager.

\textbf{Quyền hạn chính:}

\begin{itemize}
    \item Xem hàng đợi tài liệu chờ review (queue), filter theo
    repository / template / severity / submission date.

    \item Mở tài liệu kèm grounding evidence, source file, code entity,
    commit, diff, ticket, change log.

    \item So sánh AI version với approved version; kiểm tra khớp template
    và khớp source code.

    \item Comment, yêu cầu chỉnh sửa, sửa lỗi nhỏ trực tiếp.

    \item Approve (đẩy lên Manager) / Reject / trả lại Developer.
\end{itemize}

\textbf{Phạm vi dữ liệu:} đọc toàn bộ context kỹ thuật
(code, diff, commit) của tài liệu đang review, nhưng không có quyền
publish/commit cuối cùng -- đó là quyền của Manager.

\textbf{Đặc điểm quan trọng:} là ``cổng lọc kỹ thuật'' -- chặn các tài liệu
sai trước khi tốn thời gian Manager.


\subsubsection{Manager}

\textbf{Mục tiêu:} phê duyệt cuối cùng, chịu trách nhiệm publish/commit
vào repository.

\textbf{Quyền hạn chính:}

\begin{itemize}
    \item Xem tài liệu đã qua Staff review, xem staff comments,
    grounding evidence, audit trail.

    \item \textbf{Approve để publish/commit}, \textbf{Reject},
    \textbf{Rollback}.

    \item Cấu hình: loại tài liệu nào bắt buộc cần Manager approval,
    merge policy, kiểm soát publish/commit.

    \item Xem lịch sử approval; \textbf{delegate quyền approval}
    cho người khác.
\end{itemize}

\textbf{Phạm vi dữ liệu:} toàn bộ audit trail + kết quả review của Staff,
không cần xem lại chi tiết code diff (đã được Staff xử lý), tập trung vào
quyết định go/no-go.

\textbf{Đặc điểm quan trọng:} là actor \textbf{duy nhất} có quyền
publish/commit tài liệu vào repository thật -- đúng nguyên tắc
``AI không tự ý thay đổi tài liệu mà không kiểm soát''.


\subsubsection{Technical Lead}

\textbf{Mục tiêu:} quản trị (governance) toàn bộ hệ thống tài liệu ở tầm
chiến lược, không tham gia review từng tài liệu đơn lẻ.

\textbf{Quyền hạn chính:}

\begin{itemize}
    \item Quản lý documentation collection, phân quyền ownership.

    \item Cấu hình \textbf{drift detection policy} và \textbf{threshold}.

    \item Xem documentation health dashboard, dependency graph,
    quality metrics, theo dõi unresolved drift, xuất báo cáo.

    \item Cấu hình auto-update (repo nào, module nào, loại tài liệu nào),
    xem version history/change log, rollback ở cấp hệ thống.

    \item Quản lý Template: tạo/sửa/clone/xóa, định nghĩa section/heading/
    placeholder, mapping placeholder $\leftrightarrow$ code entity.
\end{itemize}

\textbf{Phạm vi dữ liệu:} toàn bộ workspace/repo thuộc phạm vi quản lý --
nhìn theo \textbf{chỉ số tổng hợp} (dashboard, metrics) hơn là từng tài liệu.

\textbf{Đặc điểm quan trọng:} khác biệt với Manager ở chỗ Manager phê duyệt
\textbf{từng tài liệu}, còn Tech Lead cấu hình \textbf{chính sách/threshold
áp dụng cho toàn hệ thống} (drift policy, template chuẩn).


\subsubsection{Administrator}

\textbf{Mục tiêu:} vận hành hạ tầng \& tích hợp toàn hệ thống.

\textbf{Quyền hạn chính:} quản lý user, role, permission, workspace,
repository; cấu hình GitHub App/OAuth/GitHub Actions; tích hợp
GitLab/Jira/Slack; cấu hình supported programming languages; cấu hình AI
settings (LLM, embedding model, confidence threshold, token budget,
indexing, retrieval); notification; theo dõi system health, AI processing
jobs, event pipeline, audit log, system analytics.

\textbf{Phạm vi dữ liệu:} toàn hệ thống, cấp hạ tầng/kỹ thuật --
\textbf{không} liên quan trực tiếp đến nội dung nghiệp vụ tài liệu
(không review/approve tài liệu).

\textbf{Đặc điểm quan trọng:} là actor duy nhất chạm vào
\textbf{cấu hình AI model} (LLM, embedding, threshold) -- tách biệt
hoàn toàn khỏi luồng nghiệp vụ review/approval.


\subsection{Chi tiết các Thực thể Dữ liệu Cốt lõi (Core Data Entities)}
\label{sec:core-data-entities}


\subsubsection{Nhóm Quản trị \& Kết nối}

\begin{itemize}

    \item \textbf{User (Thực thể mạnh):}
    Người dùng cuối trong hệ thống (Developer, Staff, Manager,
    Tech Lead, Admin).

    \item \textbf{Workspace (Thực thể mạnh):}
    Không gian làm việc chứa repository và tài liệu của một nhóm/dự án.

    \item \textbf{Role (Thực thể mạnh):}
    Vai trò của người dùng (scoped theo Workspace).

    \item \textbf{Permission (Thực thể mạnh):}
    Quyền hạn chi tiết trên hệ thống
    (VD: DOC\_APPROVE, TEMPLATE\_EDIT).
    \item\textbf{RolePermission(Thực thể mối kết hợp):}
Ma trận phân quyền chi tiết thiết lập các quyền hạn cụ thể mà một vai trò sở hữu.
    \item \textbf{WorkspaceMember (Thực thể mối kết hợp):}
    Ghi nhận sự tham gia của User trong Workspace và vai trò (Role)
    tương ứng.

    \item \textbf{GitHubConnection (Thực thể phụ thuộc):}
    Lưu thông tin xác thực OAuth của User với GitHub.

    

\end{itemize}


\subsubsection{Nhóm Mã nguồn \& Cấu trúc Code}

\begin{itemize}

    \item \textbf{Repository (Thực thể mạnh):}
    Kho chứa mã nguồn được đồng bộ vào hệ thống.

    \item \textbf{Branch (Thực thể yếu -- phụ thuộc Repository):}
    Nhánh phát triển trong repository.

    \item \textbf{Commit (Thực thể yếu -- phụ thuộc Branch/Repo):}
    Bản ghi một lần commit mã nguồn.

    \item \textbf{PullRequest (Thực thể yếu -- phụ thuộc Repository):}
    Yêu cầu hợp nhất nhánh (PR) từ GitHub.

    \item \textbf{CodeDiff (Thực thể yếu -- phụ thuộc Commit/PR):}
    Chi tiết thay đổi mã nguồn trong commit hoặc PR.

    \item \textbf{File (Thực thể yếu -- phụ thuộc Repository):}
    Cấu trúc tập tin mã nguồn trong kho chứa.

    \item \textbf{CodeEntity (Thực thể yếu -- phụ thuộc File):}
    Đơn vị mã nguồn cụ thể thu được từ AST
    (Class, Method, Function, \ldots).

    \item \textbf{Dependency (Thực thể mối kết hợp):}
    Mối quan hệ phụ thuộc giữa các CodeEntity.

\end{itemize}


\subsubsection{Nhóm Khuôn mẫu Tài liệu (Template Engine Domain)}

\begin{itemize}

    \item \textbf{Template (Thực thể mạnh):}
    Khuôn mẫu chuẩn để AI/User sinh tài liệu.

    \item \textbf{TemplateSection (Thực thể yếu -- phụ thuộc Template):}
    Mục con/heading trong cấu trúc dàn ý của Template.

    \item \textbf{Placeholder (Thực thể yếu -- phụ thuộc TemplateSection):}
    Biến/điểm chèn trong Section, làm nhiệm vụ nhận dữ liệu từ code.

    \item \textbf{TemplateMapping (Thực thể mối kết hợp):}
    Ánh xạ trực tiếp giữa Placeholder và CodeEntity để AI trích xuất thông tin.

\end{itemize}


\subsubsection{Nhóm Tài liệu \& Vòng đời Dữ liệu (Documentation Domain)}

\begin{itemize}

    \item \textbf{Document (Thực thể mạnh):}
    Tài liệu gốc (README, API Reference, Architecture Guide, ADR, \ldots).

    \item \textbf{DocumentVersion (Thực thể yếu -- phụ thuộc Documentation):}
    Bản ghi nội dung tài liệu theo từng mốc thời gian và trạng thái
    (Draft, In Review, Approved, Published).

    \item \textbf{ChangeLog (Thực thể yếu -- phụ thuộc DocumentVersion):}
    Lịch sử vết thay đổi chi tiết giữa các phiên bản tài liệu.

    

\end{itemize}


\subsubsection{Nhóm Kiểm duyệt \& Phê duyệt (Review \& Governance Domain)}

\begin{itemize}

    \item \textbf{Review (Thực thể mối kết hợp/Sự kiện):}
    Tiến trình Staff thực hiện đánh giá kỹ thuật trên một DocumentVersion.

    \item \textbf{ReviewComment (Thực thể yếu -- phụ thuộc Review):}
    Phản hồi, góp ý chi tiết trong quá trình kiểm duyệt.

    \item \textbf{Approval (Thực thể mối kết hợp/Sự kiện):}
    Quyết định phê duyệt xuất bản/commit tài liệu chính thức từ Manager.

\end{itemize}


\subsubsection{Nhóm AI Evidence, Giám sát \& Kiểm toán
